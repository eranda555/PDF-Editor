cmake_minimum_required(VERSION 3.22)
project(pdf_native_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND DEFINED PDF_PREBUILT_ROOT)
  # The NDK restricts package, header, and library searches to target roots.
  get_filename_component(pdf_android_prefix "${PDF_PREBUILT_ROOT}/${ANDROID_ABI}" ABSOLUTE)
  list(PREPEND CMAKE_FIND_ROOT_PATH "${pdf_android_prefix}")
endif()
find_package(qpdf 12.4 CONFIG REQUIRED)
find_path(PDFIUM_INCLUDE_DIR fpdfview.h PATH_SUFFIXES include public include/pdfium REQUIRED)
find_library(PDFIUM_LIBRARY NAMES pdfium REQUIRED)
add_library(pdfium_runtime SHARED IMPORTED)
set_target_properties(pdfium_runtime PROPERTIES IMPORTED_LOCATION "${PDFIUM_LIBRARY}")
add_library(pdf_native_bridge SHARED pdf_native_bridge.cpp)
target_include_directories(pdf_native_bridge PRIVATE ${PDFIUM_INCLUDE_DIR})
target_link_libraries(pdf_native_bridge PRIVATE qpdf::libqpdf pdfium_runtime)
target_compile_definitions(pdf_native_bridge PRIVATE PDF_BRIDGE_EXPORTS)
if(MSVC)
  target_compile_options(pdf_native_bridge PRIVATE /W4 /EHsc)
else()
  target_compile_options(pdf_native_bridge PRIVATE -Wall -Wextra -Werror=return-type -fvisibility=hidden)
endif()
install(TARGETS pdf_native_bridge DESTINATION lib)
